    <script>
        // Initialize Lucide Icons
        lucide.createIcons();

        // --- Theme Toggling ---
        const html = document.documentElement;
        const themeToggle = document.getElementById('theme-toggle');
        const themeIcon = document.getElementById('theme-icon');

        // Apply saved theme on load
        if (localStorage.getItem('theme') === 'dark') {
            html.classList.add('dark');
            themeIcon.setAttribute('data-lucide', 'sun');
        } else {
            themeIcon.setAttribute('data-lucide', 'moon');
        }
        lucide.createIcons();

        themeToggle.addEventListener('click', () => {
            html.classList.toggle('dark');
            if (html.classList.contains('dark')) {
                localStorage.setItem('theme', 'dark');
                themeIcon.setAttribute('data-lucide', 'sun');
            } else {
                localStorage.setItem('theme', 'light');
                themeIcon.setAttribute('data-lucide', 'moon');
            }
            lucide.createIcons();
            renderCharts(); // Re-render charts to apply new theme colors
        });

        // --- Bottom Navigation ---
        const navOverview = document.getElementById('nav-overview');
        const navProfile = document.getElementById('nav-profile');
        const viewOverview = document.getElementById('view-overview');
        const viewProfile = document.getElementById('view-profile');

        navOverview.addEventListener('click', () => {
            viewOverview.classList.remove('hidden');
            viewProfile.classList.add('hidden');
            navOverview.classList.add('active', 'text-gold');
            navOverview.classList.remove('text-secondary');
            navProfile.classList.remove('active', 'text-gold');
            navProfile.classList.add('text-secondary');
        });

        navProfile.addEventListener('click', () => {
            viewProfile.classList.remove('hidden');
            viewOverview.classList.add('hidden');
            navProfile.classList.add('active', 'text-gold');
            navProfile.classList.remove('text-secondary');
            navOverview.classList.remove('active', 'text-gold');
            navOverview.classList.add('text-secondary');
        });

        // --- Availability Toggle ---
        const toggleSwitch = document.getElementById('availability-toggle');
        const toggleDot = toggleSwitch.querySelector('.toggle-dot');
        const statusText = document.getElementById('status-text');

        toggleSwitch.addEventListener('click', () => {
            toggleSwitch.classList.toggle('active');
            toggleDot.classList.toggle('active');
            if (toggleSwitch.classList.contains('active')) {
                statusText.textContent = 'Online & Ready';
                statusText.classList.add('text-gold');
                statusText.classList.remove('text-secondary');
            } else {
                statusText.textContent = 'Offline';
                statusText.classList.remove('text-gold');
                statusText.classList.add('text-secondary');
            }
        });

        // --- Copy to Clipboard ---
        const copyBtn = document.getElementById('copy-btn');
        const copyIcon = document.getElementById('copy-icon');
        const copyText = document.getElementById('copy-text');
        const genericNumber = document.getElementById('generic-number').innerText;

        copyBtn.addEventListener('click', () => {
            navigator.clipboard.writeText(genericNumber).then(() => {
                copyIcon.setAttribute('data-lucide', 'check');
                copyText.textContent = 'Copied';
                lucide.createIcons();
                
                setTimeout(() => {
                    copyIcon.setAttribute('data-lucide', 'copy');
                    copyText.textContent = 'Copy';
                    lucide.createIcons();
                }, 2000);
            });
        });

        // --- Rate Increment/Decrement ---
        const rateMinus = document.getElementById('rate-minus');
        const ratePlus = document.getElementById('rate-plus');
        const rateDisplay = document.getElementById('rate-display');
        const rateProfileDisplay = document.getElementById('rate-display-profile');
        
        let currentRate = 4.00;

        const updateRateDisplay = () => {
            const formattedRate = `$${currentRate.toFixed(2)}`;
            rateDisplay.textContent = formattedRate;
            rateProfileDisplay.textContent = formattedRate;
        };

        rateMinus.addEventListener('click', () => {
            if (currentRate > 1.00) {
                currentRate -= 1.00;
                updateRateDisplay();
            }
        });

        ratePlus.addEventListener('click', () => {
            if (currentRate < 20.00) {
                currentRate += 1.00;
                updateRateDisplay();
            }
        });

        // --- History Tabs ---
        const tabCalls = document.getElementById('tab-calls');
        const tabPayouts = document.getElementById('tab-payouts');
        const listCalls = document.getElementById('list-calls');
        const listPayouts = document.getElementById('list-payouts');

        function switchTab(activeTab, inactiveTab, activeList, inactiveList) {
            activeTab.classList.add('text-gold', 'border-gold');
            activeTab.classList.remove('text-secondary', 'border-transparent');
            inactiveTab.classList.remove('text-gold', 'border-gold');
            inactiveTab.classList.add('text-secondary', 'border-transparent');
            
            activeList.classList.remove('hidden');
            inactiveList.classList.add('hidden');
        }

        tabCalls.addEventListener('click', () => switchTab(tabCalls, tabPayouts, listCalls, listPayouts));
        tabPayouts.addEventListener('click', () => switchTab(tabPayouts, tabCalls, listPayouts, listCalls));

        // --- Charts (Chart.js) ---
        let barChart, pieChart;

        function getChartColors() {
            const isDark = html.classList.contains('dark');
            return {
                gold: '#D4AF37',
                red: '#EF4444',
                grey: isDark ? '#4A4A4A' : '#E5E7EB',
                textColor: isDark ? '#9CA3AF' : '#6B7280',
                gridColor: isDark ? '#2D2D2D' : '#F3F4F6'
            };
        }

        function renderCharts() {
            const colors = getChartColors();
            const ctxBar = document.getElementById('barChart').getContext('2d');
            const ctxPie = document.getElementById('pieChart').getContext('2d');

            if (barChart) barChart.destroy();
            if (pieChart) pieChart.destroy();

            // Bar Chart
            barChart = new Chart(ctxBar, {
                type: 'bar',
                data: {
                    labels: ['Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat', 'Sun'],
                    datasets: [{
                        label: 'Earnings ($)',
                        data: [120, 150, 80, 200, 340, 450, 300],
                        backgroundColor: colors.gold,
                        borderRadius: 4,
                        barPercentage: 0.6
                    }]
                },
                options: {
                    responsive: true,
                    maintainAspectRatio: false,
                    plugins: { legend: { display: false } },
                    scales: {
                        y: {
                            grid: { color: colors.gridColor, drawBorder: false },
                            ticks: { color: colors.textColor, callback: (value) => `$${value}` }
                        },
                        x: {
                            grid: { display: false },
                            ticks: { color: colors.textColor }
                        }
                    }
                }
            });

            // Pie Chart
            pieChart = new Chart(ctxPie, {
                type: 'doughnut',
                data: {
                    labels: ['Answered', 'Declined', 'Missed'],
                    datasets: [{
